default_party_popularity_PP_gain = {
	clear_variable = political_power_ideology
	clr_country_flag = party_popularity_pp_gain_disabled
	hidden_effect = {
		if = {
			limit = {
				has_dynamic_modifier = { modifier = party_popularity_dynamic_modifier_ideology_mapped }
			}
			remove_dynamic_modifier = { modifier = party_popularity_dynamic_modifier_ideology_mapped }
		}
		if = {
			limit = {
				NOT = { has_dynamic_modifier = { modifier = party_popularity_dynamic_modifier } }
			}
			add_dynamic_modifier = { modifier = party_popularity_dynamic_modifier }
		}
	}
	update_coalition_pp_gain = yes
}

	clr_country_flag = party_popularity_pp_gain_disabled

		}

	remove_from_coalition = yes
	update_coalition_pp_gain = yes

	set_country_flag = party_popularity_pp_gain_disabled
	clear_variable = political_power_ideology

	hidden_effect = {
		if = {

	update_coalition_pp_gain = yes

# Updates ruling_party_pp_gain, coalition_pp_gain, and Fuhrerreich's coalition_ruling_combined_pop
update_coalition_pp_gain = {
	if = {
		limit = { has_country_flag = party_popularity_pp_gain_disabled }
		set_variable = { ruling_party_pp_gain = 0 }
		set_variable = { coalition_ruling_combined_pop = party_popularity@ruling_party }
	}
	else_if = {
		limit = { has_variable = political_power_ideology }
		set_variable = { ruling_party_pp_gain = party_popularity@var:political_power_ideology }
		set_variable = { coalition_ruling_combined_pop = ruling_party_pp_gain }
	}
	else = {
		set_variable = { ruling_party_pp_gain = party_popularity@ruling_party }
		set_variable = { coalition_ruling_combined_pop = ruling_party_pp_gain }
	}

	if = {
		limit = { NOT = { has_variable = coalition_stability } }
		set_variable = { coalition_stability = 0.5 }
	}

	if = {
		limit = {
			check_variable = { coalition_partners^num > 0 }
		}
		set_variable = { coalition_pp_gain = 0 }
		for_each_loop = {
			array = coalition_partners
			add_to_variable = { coalition_pp_gain = party_popularity@var:v }
			add_to_variable = { coalition_ruling_combined_pop = party_popularity@var:v }
		}
		multiply_variable = { coalition_pp_gain = coalition_stability }
	}
	else = {
		clear_variable = coalition_pp_gain
		hidden_effect = {
			if = {
				limit = {
					has_dynamic_modifier = { modifier = party_popularity_dynamic_modifier_coalition }
				}
				remove_dynamic_modifier = { modifier = party_popularity_dynamic_modifier_coalition }
			}
		}
	}
}

add_to_coalition = {
	# Ensure base PP dynamic modifier is active even if called from history files before on_startup
	if = {
		limit = {
			NOT = { has_country_flag = party_popularity_pp_gain_disabled }
			NOT = { has_dynamic_modifier = { modifier = party_popularity_dynamic_modifier } }
			NOT = { has_dynamic_modifier = { modifier = party_popularity_dynamic_modifier_ideology_mapped } }
		}
		default_party_popularity_PP_gain = yes
	}
	if = {
		limit = {
			NOT = {
				AND = {
					has_variable = political_power_ideology
					check_variable = { political_power_ideology = coalition_partner_var }
				}
				is_in_array = { coalition_partners = coalition_partner_var }
				AND = {
					has_government = var:coalition_partner_var
					has_dynamic_modifier = { modifier = party_popularity_dynamic_modifier }
				}
			}
		}
		custom_effect_tooltip = add_to_coalition
		set_variable = { is_in_coalition_with_@var:coalition_partner_var = 1 }
		add_to_array = { coalition_partners = coalition_partner_var }
		hidden_effect = {
			if = {
				limit = {
					NOT = { has_dynamic_modifier = { modifier = party_popularity_dynamic_modifier_coalition } }
				}
				add_dynamic_modifier = { modifier = party_popularity_dynamic_modifier_coalition }
			}
		}
		update_coalition_pp_gain = yes
	}
}

remove_from_coalition = {
	if = {
		limit = {
			is_in_array = { coalition_partners = coalition_partner_var }
		}
		custom_effect_tooltip = remove_from_coalition
		hidden_effect = {
			clear_variable = is_in_coalition_with_@var:coalition_partner_var
			remove_from_array = { coalition_partners = coalition_partner_var }
			if = {
				limit = {
					check_variable = { coalition_partners^num = 0 }
				}
				clear_array = coalition_partners
			}
			update_coalition_pp_gain = yes
		}
	}
}

end_coalition = {
	hidden_effect = {
		if = {
			limit = {
				has_dynamic_modifier = { modifier = party_popularity_dynamic_modifier_coalition }
			}
			remove_dynamic_modifier = { modifier = party_popularity_dynamic_modifier_coalition }
		}
	}
	clear_variable = coalition_pp_gain
	clear_array = coalition_partners

	clear_variable = is_in_coalition_with_@token:communism
	clear_variable = is_in_coalition_with_@token:socialism
	clear_variable = is_in_coalition_with_@token:social_democracy
	clear_variable = is_in_coalition_with_@token:progressivism
	clear_variable = is_in_coalition_with_@token:centrism
	clear_variable = is_in_coalition_with_@token:liberalism
	clear_variable = is_in_coalition_with_@token:conservatism
	clear_variable = is_in_coalition_with_@token:national_democracy
	clear_variable = is_in_coalition_with_@token:traditionalism
	clear_variable = is_in_coalition_with_@token:authoritarianism
	clear_variable = is_in_coalition_with_@token:ultranationalism
	clear_variable = is_in_coalition_with_@token:esotericism
	clear_variable = is_in_coalition_with_@token:oligarchism

	update_coalition_pp_gain = yes
}

change_ruling_party_popularity = {
	if = {
		limit = {
			OR = {
				check_variable = { ideology_change_var > 1 }
				check_variable = { ideology_change_var < -1 }
			}
		}
		multiply_temp_variable = { ideology_change_var = 0.01 }
	}
	add_popularity = {
		ideology = var:current_party_ideology_group
		popularity = ideology_change_var
	}
	update_coalition_pp_gain = yes
}
