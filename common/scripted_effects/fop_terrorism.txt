fop_storm_terrorists = {
	# Calculate agency strength
	set_temp_variable = { agency_power = 0 }

	# Base for having an agency
	if = {
		limit = { has_intelligence_agency = yes }
		add_to_temp_variable = { agency_power = 4 }
	}

	# Strong bonuses from upgrades
	if = { limit = { has_done_agency_upgrade = upgrade_passive_defense }		add_to_temp_variable = { agency_power = 12 } }
	if = { limit = { has_done_agency_upgrade = upgrade_anti_partisan }			add_to_temp_variable = { agency_power = 14 } }
	if = { limit = { has_done_agency_upgrade = upgrade_economy_civilian }		add_to_temp_variable = { agency_power = 3 } }
	if = { limit = { has_done_agency_upgrade = upgrade_invisible_ink }			add_to_temp_variable = { agency_power = 5 } }
	if = { limit = { has_done_agency_upgrade = upgrade_training_centers }		add_to_temp_variable = { agency_power = 12 } }
	if = { limit = { has_done_agency_upgrade = upgrade_isolation_training }		add_to_temp_variable = { agency_power = 5 } }
	if = { limit = { has_done_agency_upgrade = upgrade_psychological_warfare }	add_to_temp_variable = { agency_power = 12 } }
	if = { limit = { has_done_agency_upgrade = upgrade_suicide_pills }			add_to_temp_variable = { agency_power = 3 } }
	if = { limit = { has_done_agency_upgrade = upgrade_diplo_training }			add_to_temp_variable = { agency_power = 8 } }
	if = { limit = { has_done_agency_upgrade = upgrade_naval_department }		add_to_temp_variable = { agency_power = 3 } }
	if = { limit = { has_done_agency_upgrade = upgrade_air_department }			add_to_temp_variable = { agency_power = 3 } }
	if = { limit = { has_done_agency_upgrade = upgrade_tech_department }		add_to_temp_variable = { agency_power = 5 } }
	if = { limit = { has_done_agency_upgrade = upgrade_crypto_department }		add_to_temp_variable = { agency_power = 5 } }
	if = { limit = { has_done_agency_upgrade = upgrade_defense_department }		add_to_temp_variable = { agency_power = 10 } }

	# Operatives are also very important
	set_temp_variable = { op_bonus = num_of_operatives }
	multiply_temp_variable = { op_bonus = 2 }
	add_to_temp_variable = { agency_power = op_bonus }

	# Maximum cap
	clamp_temp_variable = {
		var = agency_power
		min = 0
		max = 100
	}

	random_list = {
		# 1. Full Success
		10 = {
			modifier = {
				add = agency_power
				factor = 5
			}
			country_event = { id = fop_terrorism.2 hours = 2 }
		}

		# 2. Partial Success
		20 = {
			modifier = {
				add = agency_power
				factor = 1.3
			}
			country_event = { id = fop_terrorism.3 hours = 2 }
		}

		# 3. Brutal Problem
		35 = {
			modifier = {
				factor = 1.5
				subtract = agency_power
				add = 10
			}
			country_event = { id = fop_terrorism.4 hours = 2 }
		}

		# 4. Full Failure – terrorists escaped
		35 = {
			modifier = {
				factor = 1.9
				subtract = agency_power
				add = 14
			}
			country_event = { id = fop_terrorism.5 hours = 2 }
		}
	}
}
