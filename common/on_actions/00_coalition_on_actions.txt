# common/on_actions/00_coalition_on_actions.txt

on_actions = {
    on_startup = {
        effect = {
            every_country = {
                if = {
                    limit = { has_variable = political_power_ideology }
                    map_party_popularity_PP_gain_to_ideology = yes
                }
                else = {
                    default_party_popularity_PP_gain = yes
                }
                update_coalition_pp_gain = yes
            }
        }
    }

    on_daily = {
        effect = {
            if = {
                limit = { has_variable = coalition_partners^num }
                update_coalition_pp_gain = yes
            }
        }
    }

    on_ruling_party_change = {
        effect = {
            # Prevent the new ruling ideology from also counting as a coalition partner
            if = {
                limit = { is_in_array = { coalition_partners = current_party_ideology_group } }
                remove_from_array = { array = coalition_partners value = current_party_ideology_group }
                if = {
                    limit = { check_variable = { coalition_partners^num = 0 } }
                    clear_array = coalition_partners
                }
                update_coalition_pp_gain = yes
            }
        }
    }
}
