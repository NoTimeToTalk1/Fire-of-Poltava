country_event = {
	id = fop_terrorism.1
	title = fop_terrorism.1.t
	desc = fop_terrorism.1.d
	picture = [GetTerrorEventPic]

	is_triggered_only = yes


	immediate = {
		hidden_effect = {
		
		set_variable = { fop_terror_pic = random }
		multiply_variable = { fop_terror_pic = 3 }
		round_variable = fop_terror_pic
		add_to_variable = { fop_terror_pic = 1 }

		random_owned_controlled_state = {
			limit = { is_capital = no }
			save_event_target_as = terror_state
		}
		if = {
			limit = { NOT = { has_event_target = terror_state } }
			random_owned_controlled_state = {
				save_event_target_as = terror_state
			}
		}
		set_variable = { fop_terror_hour = random }
		multiply_variable = { fop_terror_hour = 23 }
		round_variable = fop_terror_hour
		clamp_variable = {
			var = fop_terror_hour
			min = 0
			max = 23
		}
		random_list = {
			10 = { set_variable = { fop_terror_place = 1 } }
			10 = { set_variable = { fop_terror_place = 2 } }
			8  = { set_variable = { fop_terror_place = 3 } }
			8  = { set_variable = { fop_terror_place = 4 } }
			7  = { set_variable = { fop_terror_place = 5 } }
			6  = { set_variable = { fop_terror_place = 6 } }
			5  = { set_variable = { fop_terror_place = 7 } }
			5  = { set_variable = { fop_terror_place = 8 } }
			4  = { set_variable = { fop_terror_place = 9 } }
			3  = { set_variable = { fop_terror_place = 10 } }
		}
		fop_select_terror_attack = yes
		}
	}

	option = {
		name = fop_terrorism.1.a
		ai_chance = { factor = 40 }

		fop_apply_terror_demand = yes
		custom_effect_tooltip = fop_terrorism_surrender_tt
	}

	option = {
		name = fop_terrorism.1.b
		ai_chance = { factor = 60 }

		fop_calculate_agency_power = yes
		custom_effect_tooltip = [GetStormChanceTooltip]

		hidden_effect = {
		fop_storm_terrorists = yes
		}
	}
}
